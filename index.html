const input = document.getElementById("url"); const button = document.getElementById("open"); function openPage() { let text = input.value.trim(); if (text === "") { return; } let address; if ( text.startsWith("http://") || text.startsWith("https://") ) { address = text; } else if ( text.includes(".") && !text.includes(" ") ) { address = "https://" + text; } else { address = "https://www.google.com/search?q=" + encodeURIComponent(text); } chrome.tabs.update({ url: address }); } button.addEventListener("click", openPage); input.addEventListener("keydown", function(event) { if (event.key === "Enter") { openPage(); } });
